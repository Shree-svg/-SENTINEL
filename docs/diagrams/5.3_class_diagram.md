# 5.3 Class Diagram

## SENTINEL: Key Classes, Interfaces & Relationships

```mermaid
classDiagram
    %% ─── TYPE INTERFACES ───────────────────────────────────────────────

    class Reading {
        +string sensorId
        +number raw
        +number gas
        +number temperature
        +number temp
        +number humidity
        +number rh
        +number seq
        +boolean alarm
        +number? compensatedGas
        +number? steamEstimate
        +number? slope
        +number? timeToThresholdSeconds
        +string deviceTime
        +string serverTime
        +SensorHealth? sensorHealth
        +Classification? classification
    }

    class SensorHealth {
        +boolean isDegraded
        +string status
        +string[] warnings
        +number? warmupRemainingSeconds
        +boolean? flatlined
        +boolean? dhtFailed
    }

    class Classification {
        +string? readingId
        +ClassificationLabel label
        +number confidence
        +string[] factors
        +Severity severity
        +boolean? isLowConfidenceOverride
    }

    class Alert {
        +string _id
        +string? classificationId
        +Classification classification
        +ReadingSnapshot? readingSnapshot
        +string channel
        +AlertStatus status
        +string createdAt
        +string? acknowledgedAt
        +number? escalationCount
        +boolean? isSimulated
        +boolean? isDemo
    }

    class CaregiverLink {
        +string _id
        +string residentId
        +string residentName
        +string residentAddress
        +string caregiverId
        +string caregiverName
        +string caregiverPhone
        +string relationship
        +boolean active
        +string linkedAt
    }

    class NotificationSettings {
        +boolean browserNotificationsEnabled
        +boolean soundAlertsEnabled
        +number soundVolume
        +number escalationIntervalSeconds
        +boolean emailEscalationEnabled
        +boolean smsEscalationEnabled
        +boolean caregiverPushEnabled
        +boolean suppressInterferenceAlerts
        +boolean dndEnabled
        +string dndStartTime
        +string dndEndTime
        +boolean dndBypassForGenuineLeak
    }

    %% ─── PIPELINE CONFIGURATION ────────────────────────────────────────

    class PipelineConfig {
        <<constant>>
        +number NAIVE_THRESHOLD = 300
        +number WARMUP_SAMPLES = 10
        +number K_RH = 2.8
        +number K_T = 1.2
        +number MAX_STEAM_DEDUCTION = 250
        +number DANGER_THRESHOLD = 500
        +number WARN_RESIDUAL = 60
        +number ALARM_RESIDUAL = 150
        +number DANGER_RESIDUAL = 300
        +number WINDOW = 10
        +number ALARM_CONFIRM = 3
        +number ESCALATE_AFTER = 15
        +number EMERGENCY_AFTER = 30
    }

    class Analysis {
        +Cls cls
        +Level level
        +boolean calibrating
        +number calibProgress
        +number baseline
        +number compensated
        +number residual
        +number slope
        +number rhSlope
        +number? timeToDanger
        +boolean naiveAlarm
        +boolean smartAlarm
        +boolean saturated
        +number alarmTicks
        +string[] events
    }

    class Counters {
        +number naiveFalse
        +number naiveTrue
        +number smartFalse
        +number smartTrue
    }

    class Pipeline {
        -number[] rawBuf
        -number[] rhBuf
        -number[] resBuf
        -CalibrationState cal
        -BaselineState? base
        -number confirm
        -boolean latched
        -number alarmTicks
        -boolean acked
        -Level level
        -Cls cls
        -Counters counters
        +process(Reading, truth) Analysis
        +ack() void
    }

    %% ─── REACT HOOKS / SERVICES ─────────────────────────────────────────

    class usePipeline {
        <<hook>>
        -Pipeline pipe
        +Analysis? analysis
        +Counters counters
        +function ack
        +push(Reading) void
    }

    class useDeviceConnection {
        <<hook>>
        +Reading? latest
        +Reading[] history
        +boolean connected
        +string? port
        +boolean demoActive
        +string scenario
        +startDemo() void
        +stopDemo() void
        +setScenario(name) void
    }

    class usePolling {
        <<hook>>
        +startPolling() void
        +stopPolling() void
    }

    class NotificationService {
        <<service>>
        +requestPermission() void
        +notify(title, body) void
        +playAlertSound() void
    }

    %% ─── BACKEND / SERVER ───────────────────────────────────────────────

    class IngestionBridge {
        <<server>>
        -SerialPort port
        -ReadlineParser parser
        -Reading[] buffer
        -Alert[] alerts
        -boolean demoActive
        +startWatchdog() void
        +parseFrame(raw) Reading
        +handleTelemetry(Reading) void
        +GET /api/readings/latest()
        +GET /api/readings/history()
        +GET /api/alerts()
        +POST /api/alerts/acknowledge()
        +POST /api/demo/start()
        +POST /api/demo/stop()
        +POST /api/demo/scenario()
        +GET /api/hardware/status()
    }

    class DemoSimulator {
        <<server>>
        -string scenario
        -boolean running
        -Timer intervalId
        +start(scenario) void
        +stop() void
        +setScenario(name) void
        +generateFrame() Reading
        +getStatus() object
    }

    %% ─── FIRMWARE (C++ → represented as class) ──────────────────────────

    class SentinelFirmware {
        <<firmware>>
        -int MQ2_PIN = A0
        -int DHTPIN = 2
        -int BUZZER_PIN = 8
        -int LED_PIN = 9
        -int BT_RX = 10
        -int BT_TX = 11
        -int threshold = 300
        -uint32_t WARMUP_TIME_MS = 30000
        -DHT dht
        -SoftwareSerial BT
        -uint16_t seq
        +setup() void
        +loop() void
        +sendPacket(gas, temp, hum, alarm) void
    }

    %% ─── RELATIONSHIPS ──────────────────────────────────────────────────

    Reading "1" --> "0..1" SensorHealth : has
    Reading "1" --> "0..1" Classification : has
    Alert "1" --> "1" Classification : embeds
    Classification --> Alert : generates

    Pipeline --> Analysis : produces
    Pipeline --> Counters : tracks
    Pipeline ..|> PipelineConfig : configured by

    usePipeline --> Pipeline : owns
    usePipeline --> Analysis : exposes
    useDeviceConnection --> Reading : streams
    usePipeline ..> Reading : consumes

    IngestionBridge --> DemoSimulator : delegates demo
    IngestionBridge --> Reading : emits
    IngestionBridge --> Alert : stores
    IngestionBridge --> NotificationSettings : reads
    SentinelFirmware --> IngestionBridge : serial→USB

    CaregiverLink --> Alert : notified by
    NotificationSettings --> IngestionBridge : configures dispatch
```
